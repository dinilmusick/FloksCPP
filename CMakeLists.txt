cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/version.txt" ULTRST_RELEASE_VERSION LIMIT_COUNT 1)
project(FloksCPP VERSION ${ULTRST_RELEASE_VERSION} LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
option(FLOKSCPP_STRICT_WARNINGS "Treat warnings as errors" ON)
option(FLOKSCPP_SANITIZERS "Enable ASan and UBSan" OFF)
option(FLOKSCPP_REQUIRE_FRNL "Fail configuration unless exact installed FrnlCPP package exists" OFF)
add_library(floks_core INTERFACE)
add_library(FloksCPP::core ALIAS floks_core)
set_target_properties(floks_core PROPERTIES EXPORT_NAME core)
target_compile_features(floks_core INTERFACE cxx_std_17)
target_include_directories(floks_core INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
if(BUILD_TESTING)
  function(floks_test name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE FloksCPP::core)
    if(MSVC)
      target_compile_options(${name} PRIVATE /UNDEBUG)
      if(FLOKSCPP_STRICT_WARNINGS)
        target_compile_options(${name} PRIVATE /W4 /WX)
      endif()
    else()
      target_compile_options(${name} PRIVATE -UNDEBUG)
      if(FLOKSCPP_STRICT_WARNINGS)
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
      endif()
      if(FLOKSCPP_SANITIZERS)
        target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${name} PRIVATE -fsanitize=address,undefined)
      endif()
    endif()
    add_test(NAME ${name} COMMAND ${name})
  endfunction()
  floks_test(floks_components_tests tests/components.cpp)
  floks_test(floks_graph_tests tests/graph.cpp)
  floks_test(floks_music_tests tests/music.cpp)
  floks_test(floks_serialization_tests tests/serialization.cpp)
  floks_test(floks_controls_tests tests/controls.cpp)
  floks_test(floks_hardening_tests tests/hardening.cpp)
  floks_test(floks_editor_tests tests/editor.cpp)
  if(FLOKSCPP_REQUIRE_FRNL)
    find_package(FrnlCPP CONFIG REQUIRED)
  else()
    find_package(FrnlCPP CONFIG QUIET)
  endif()
  if(FrnlCPP_FOUND)
    floks_test(floks_frnl_integration_tests tests/frnl.cpp)
    target_link_libraries(floks_frnl_integration_tests PRIVATE FrnlCPP::semantic)
  endif()
endif()
install(TARGETS floks_core EXPORT FloksCPPTargets)
install(DIRECTORY include/floks DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FloksCPPTargets FILE FloksCPPTargets.cmake NAMESPACE FloksCPP:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FloksCPP)
configure_package_config_file(cmake/FloksCPPConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/FloksCPPConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FloksCPP)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/FloksCPPConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/FloksCPPConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/FloksCPPConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FloksCPP)
