cmake_minimum_required(VERSION 3.16)
project(FloksConsumer LANGUAGES CXX)
include(CTest)
find_package(FloksCPP CONFIG REQUIRED)
find_package(FrnlCPP CONFIG QUIET)
add_executable(floks_consumer main.cpp)
target_link_libraries(floks_consumer PRIVATE FloksCPP::core)
if(FrnlCPP_FOUND)
  target_link_libraries(floks_consumer PRIVATE FrnlCPP::semantic)
  target_compile_definitions(floks_consumer PRIVATE FLOKSCPP_CONSUMER_FRNL=1)
endif()
add_test(NAME floks_installed_consumer COMMAND floks_consumer)
